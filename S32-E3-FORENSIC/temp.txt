# LIVE EVIDENCE LOGGER

class LiveEvidenceLogger:
    """Logger that writes JSONL events during execution, not reconstructed after.
    
    The evidence file is shared between parent and child via a temp file.
    The child writes to this file during execution, and the parent reads it
    after completion.
    """
    
    def __init__(self, filepath: str):
        self.filepath = filepath
        self._ensure_file()
    
    def _ensure_file(self):
        """Ensure the JSONL file exists and is empty for fresh run."""
        os.makedirs(os.path.dirname(self.filepath), exist_ok=True)
        # Start fresh - truncate any existing file
        with open(self.filepath, "w", encoding="utf-8") as f:
            f.write("")  # Empty fresh file
    
    def write_event(self, cycle: int, event: str, pid: int, **extra):
        """Write a live evidence event to JSONL."""
        with open(self.filepath, "a", encoding="utf-8") as f:
            event_data = {"cycle": cycle, "event": event, "pid": pid}
            event_data.update(extra)
            f.write(json.dumps(event_data, ensure_ascii=False) + "\n")
            f.flush()
    
    def read_events(self) -> List[Dict[str, Any]]:
        """Read all events from the evidence file after child completion."""
        events = []
        if not os.path.exists(self.filepath):
            return events
        
        try:
            with open(self.filepath, "r", encoding="utf-8") as f:
                for line in f:
                    line = line.strip()
                    if line:
                        try:
                            event = json.loads(line)
                            events.append(event)
                        except json.JSONDecodeError:
                            pass
        except Exception:
            pass
        
        return events
    
    def close(self):
        """Close the logger - no file to close since we overwrite each run."""
        pass

def run_child_process(cycle: int, target_path: str, data_file: str,
                      evidence_path: str) -> subprocess.Popen:
    """Run the child script as a subprocess and return the Popen object.
    
    The child writes JSONL evidence to the shared evidence file during execution.
    Uses --data-file to avoid JSON quoting issues with command-line arguments.
    The parent can monitor the evidence file and capture stdout events.
    
    Key: data_file path contains JSON like {"cycle": 1, "objective": "..."}
    """
    cmd = [
        sys.executable, CHILD_SCRIPT_PATH,
        "--target", target_path,
        "--data-file", data_file,
        "--cycle", str(cycle),
        "--evidence", evidence_path
    ]
    
    proc = subprocess.Popen(
        cmd,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        stdin=subprocess.PIPE,
        text=True,
        bufsize=1
    )
    
    return proc
